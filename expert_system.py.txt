class Rule:
    def __init__(self, conditions, conclusion):
        self.conditions = set(conditions)
        self.conclusion = conclusion


class ExpertSystem:
    def __init__(self):
        self.facts = set()
        self.rules = []
        self.inference_log = []

    def add_fact(self, fact):
        self.facts.add(fact)

    def add_rule(self, conditions, conclusion):
        self.rules.append(Rule(conditions, conclusion))

    def forward_chaining(self):
        changed = True

        while changed:
            changed = False

            for rule in self.rules:
                if (
                    rule.conditions.issubset(self.facts)
                    and rule.conclusion not in self.facts
                ):
                    self.facts.add(rule.conclusion)

                    log = (
                        f"Rule Applied: "
                        f"{' AND '.join(rule.conditions)} "
                        f"--> {rule.conclusion}"
                    )

                    self.inference_log.append(log)
                    changed = True

    def display_results(self):
        print("\n===== Inference Steps =====")
        for step in self.inference_log:
            print(step)

        print("\n===== Final Facts =====")
        for fact in sorted(self.facts):
            print("-", fact)


def main():
    system = ExpertSystem()

    # Rules
    system.add_rule(["fever", "cough"], "flu")
    system.add_rule(["flu", "body_pain"], "doctor_visit")
    system.add_rule(["sore_throat", "cough"], "cold")
    system.add_rule(["cold", "fever"], "infection")
    system.add_rule(["infection"], "medical_checkup")

    print("Rule-Based Expert System")
    print("\nEnter symptoms one by one.")
    print("Type 'done' when finished.\n")

    while True:
        symptom = input("Enter symptom: ").lower()

        if symptom == "done":
            break

        system.add_fact(symptom)

    system.forward_chaining()
    system.display_results()


if __name__ == "__main__":
    main()
